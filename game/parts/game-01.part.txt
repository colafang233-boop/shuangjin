  class Companion extends Entity {
    constructor(x, y) {
      super(x, y, 22);
      this.name = '游侠·陆沉';
      this.maxHp = 140;
      this.hp = this.maxHp;
      this.damage = 16;
      this.speed = 150;
      this.attackCooldown = 0;
      this.reviveTimer = 0;
      this.mode = 'attack';
    }
    update(game, dt) {
      if (this.dead) {
        this.reviveTimer -= dt;
        if (this.reviveTimer <= 0) {
          this.dead = false;
          this.hp = this.maxHp * 0.55;
          this.x = game.player.x + 35;
          this.y = game.player.y + 35;
          game.message('陆沉重新站了起来。', 'good');
        }
        return;
      }
      this.attackCooldown -= dt;
      let target = null;
      let best = 420;
      for (const enemy of game.enemies) {
        if (enemy.dead) continue;
        const d = distance(this, enemy);
        if (d < best) { best = d; target = enemy; }
      }
      if (target && this.mode === 'attack') {
        if (best > 58) game.moveEntity(this, target.x - this.x, target.y - this.y, this.speed, dt);
        else if (this.attackCooldown <= 0) {
          this.attackCooldown = 1.15;
          game.dealDamage(target, this.damage + game.state.player.level * 2, 'companion');
          game.addEffect({ type: 'slash', x: target.x, y: target.y, life: 0.22, color: '#f2cf86' });
        }
      } else {
        const d = distance(this, game.player);
        if (d > 90) game.moveEntity(this, game.player.x - this.x, game.player.y - this.y, this.speed, dt);
      }
    }
  }

  class Enemy extends Entity {
    constructor(x, y, type, options = {}) {
      const spec = ENEMY_TYPES[type];
      super(x, y, spec.size);
      this.type = type;
      this.name = spec.name;
      this.maxHp = Math.round(spec.hp * (options.scale || 1));
      this.hp = this.maxHp;
      this.speed = spec.speed;
      this.damage = Math.round(spec.damage * (options.scale || 1));
      this.range = spec.range;
      this.cooldown = spec.cooldown;
      this.xp = Math.round(spec.xp * (options.scale || 1));
      this.color = spec.color;
      this.armor = spec.armor || 0;
      this.ranged = spec.ranged || false;
      this.mage = spec.mage || false;
      this.elite = spec.elite || false;
      this.attackTimer = rand(0.2, 1.2);
      this.slowTimer = 0;
      this.slowFactor = 1;
      this.tag = options.tag || null;
      this.aggroRange = options.aggroRange || 470;
      this.homeX = x;
      this.homeY = y;
    }
    update(game, dt) {
      if (this.dead) return;
      this.attackTimer -= dt;
      if (this.slowTimer > 0) {
        this.slowTimer -= dt;
        if (this.slowTimer <= 0) this.slowFactor = 1;
      }
      const targets = [game.player, game.companion].filter(Boolean).filter((target) => !target.dead);
      let target = targets[0];
      if (targets.length > 1 && distance(this, targets[1]) < distance(this, targets[0]) * 0.86) target = targets[1];
      if (!target) return;
      const d = distance(this, target);
      if (d > this.aggroRange && distance(this, { x: this.homeX, y: this.homeY }) < 40) return;

      if (this.ranged) {
        if (d < 130) game.moveEntity(this, this.x - target.x, this.y - target.y, this.speed * this.slowFactor, dt);
        else if (d > this.range * 0.82) game.moveEntity(this, target.x - this.x, target.y - this.y, this.speed * this.slowFactor, dt);
        if (d <= this.range && this.attackTimer <= 0) {
          this.attackTimer = this.cooldown;
          const dx = target.x - this.x;
          const dy = target.y - this.y;
          const l = length(dx, dy);
          game.projectiles.push(new Projectile({
            x: this.x, y: this.y - 5,
            vx: dx / l * (this.mage ? 190 : 250),
            vy: dy / l * (this.mage ? 190 : 250),
            damage: this.damage,
            owner: 'enemy',
            radius: this.mage ? 13 : 7,
            color: this.mage ? '#b89fff' : '#d9f4ff',
            mage: this.mage,
          }));
          game.sound.tone(this.mage ? 260 : 180, 0.08, 'triangle', 0.018);
        }
      } else {
        if (d > this.range) game.moveEntity(this, target.x - this.x, target.y - this.y, this.speed * this.slowFactor, dt);
        else if (this.attackTimer <= 0) {
          this.attackTimer = this.cooldown;
          game.dealDamage(target, this.damage, 'enemy');
          game.addEffect({ type: 'slash', x: target.x, y: target.y, life: 0.2, color: '#ff9aa8' });
        }
      }
    }
  }
