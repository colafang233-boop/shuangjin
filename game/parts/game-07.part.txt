        if (item.kind === 'obstacle') this.drawBox(item.data);
        if (item.kind === 'seal') this.drawSeal(item.data);
        if (item.kind === 'pickup') this.drawPickup(item.data);
        if (item.kind === 'npc') this.drawActor(item.data, 'npc');
        if (item.kind === 'enemy') this.drawActor(item.data, item.data === this.boss ? 'boss' : 'enemy');
        if (item.kind === 'companion') this.drawActor(item.data, 'companion');
        if (item.kind === 'player') this.drawActor(item.data, 'player');
        if (item.kind === 'wall') this.drawWall(item.data);
      }
      this.projectiles.forEach((projectile) => this.drawProjectile(projectile));
      this.effects.forEach((effect) => this.drawEffect(effect));
      this.drawCursor();
      this.drawVignette();
    }

    drawFloor() {
      const ctx = this.ctx;
      ctx.fillStyle = this.scene.colors.fog;
      ctx.fillRect(0, 0, W, H);
      const tile = this.scene.tile;
      for (let x = 0; x < this.scene.size.w; x += tile) {
        for (let y = 0; y < this.scene.size.h; y += tile) {
          const p1 = this.worldToScreen(x, y);
          const p2 = this.worldToScreen(x + tile, y);
          const p3 = this.worldToScreen(x + tile, y + tile);
          const p4 = this.worldToScreen(x, y + tile);
          if (Math.max(p1.x, p2.x, p3.x, p4.x) < -120 || Math.min(p1.x, p2.x, p3.x, p4.x) > W + 120 || Math.max(p1.y, p2.y, p3.y, p4.y) < -80 || Math.min(p1.y, p2.y, p3.y, p4.y) > H + 80) continue;
          ctx.beginPath();
          ctx.moveTo(p1.x, p1.y); ctx.lineTo(p2.x, p2.y); ctx.lineTo(p3.x, p3.y); ctx.lineTo(p4.x, p4.y); ctx.closePath();
          ctx.fillStyle = ((x / tile + y / tile) % 2) ? this.scene.colors.floorA : this.scene.colors.floorB;
          ctx.fill();
          ctx.strokeStyle = this.scene.colors.grid;
          ctx.stroke();
        }
      }
    }

    drawExits() {
      const ctx = this.ctx;
      for (const exit of this.scene.exits) {
        const allowed = exit.condition(this);
        const p1 = this.worldToScreen(exit.x, exit.y);
        const p2 = this.worldToScreen(exit.x + exit.w, exit.y);
        const p3 = this.worldToScreen(exit.x + exit.w, exit.y + exit.h);
        const p4 = this.worldToScreen(exit.x, exit.y + exit.h);
        ctx.beginPath();
        ctx.moveTo(p1.x, p1.y); ctx.lineTo(p2.x, p2.y); ctx.lineTo(p3.x, p3.y); ctx.lineTo(p4.x, p4.y); ctx.closePath();
        ctx.fillStyle = allowed ? 'rgba(83,223,255,.14)' : 'rgba(255,97,120,.08)';
        ctx.fill();
        ctx.strokeStyle = allowed ? 'rgba(132,237,255,.48)' : 'rgba(255,125,145,.24)';
        ctx.lineWidth = 2;
        ctx.stroke();
      }
      ctx.lineWidth = 1;
    }

    drawBox(box) {
      const ctx = this.ctx;
      const h = box.height || 40;
      const a = this.worldToScreen(box.x, box.y);
      const b = this.worldToScreen(box.x + box.w, box.y);
      const c = this.worldToScreen(box.x + box.w, box.y + box.h);
      const d = this.worldToScreen(box.x, box.y + box.h);
      const at = { x: a.x, y: a.y - h }, bt = { x: b.x, y: b.y - h }, ct = { x: c.x, y: c.y - h }, dt = { x: d.x, y: d.y - h };
      ctx.beginPath(); ctx.moveTo(a.x, a.y); ctx.lineTo(b.x, b.y); ctx.lineTo(bt.x, bt.y); ctx.lineTo(at.x, at.y); ctx.closePath();
      ctx.fillStyle = box.color || '#41505c'; ctx.fill();
      ctx.beginPath(); ctx.moveTo(b.x, b.y); ctx.lineTo(c.x, c.y); ctx.lineTo(ct.x, ct.y); ctx.lineTo(bt.x, bt.y); ctx.closePath();
      ctx.fillStyle = this.shade(box.color || '#41505c', -18); ctx.fill();
      ctx.beginPath(); ctx.moveTo(at.x, at.y); ctx.lineTo(bt.x, bt.y); ctx.lineTo(ct.x, ct.y); ctx.lineTo(dt.x, dt.y); ctx.closePath();
      ctx.fillStyle = box.roof || this.shade(box.color || '#41505c', 24); ctx.fill();
      ctx.strokeStyle = 'rgba(224,247,255,.12)'; ctx.stroke();
      if (box.label) {
        const center = this.worldToScreen(box.x + box.w / 2, box.y + box.h / 2, h + 10);
        ctx.fillStyle = 'rgba(232,248,255,.74)'; ctx.font = '12px sans-serif'; ctx.textAlign = 'center'; ctx.fillText(box.label, center.x, center.y);
      }
    }

    shade(hex, amount) {
      const value = hex.replace('#', '');
      if (value.length !== 6) return hex;
      const num = parseInt(value, 16);
      const r = clamp((num >> 16) + amount, 0, 255);
      const g = clamp(((num >> 8) & 0xff) + amount, 0, 255);
      const b = clamp((num & 0xff) + amount, 0, 255);
      return `rgb(${r},${g},${b})`;
    }

    drawProp(prop) {
      const ctx = this.ctx;
      const p = this.worldToScreen(prop.x, prop.y);
      if (prop.type === 'tree') {
        ctx.fillStyle = 'rgba(0,0,0,.22)'; ctx.beginPath(); ctx.ellipse(p.x, p.y + 8, 28, 11, 0, 0, Math.PI * 2); ctx.fill();
        ctx.fillStyle = '#4c6170'; ctx.fillRect(p.x - 5, p.y - 45, 10, 48);
        ctx.fillStyle = this.scene.id === 'snowfield' ? '#d8eef2' : '#567987';
        for (let i = 0; i < 3; i += 1) { ctx.beginPath(); ctx.moveTo(p.x, p.y - 88 + i * 22); ctx.lineTo(p.x - 32 + i * 3, p.y - 38 + i * 8); ctx.lineTo(p.x + 32 - i * 3, p.y - 38 + i * 8); ctx.closePath(); ctx.fill(); }
      }
      if (prop.type === 'rock') {
        ctx.fillStyle = '#77929c'; ctx.beginPath(); ctx.moveTo(p.x - 24, p.y); ctx.lineTo(p.x - 12, p.y - 26); ctx.lineTo(p.x + 15, p.y - 31); ctx.lineTo(p.x + 28, p.y - 4); ctx.lineTo(p.x + 8, p.y + 9); ctx.closePath(); ctx.fill();
      }
      if (prop.type === 'fire' || prop.type === 'camp') {
        ctx.fillStyle = '#5d4538'; ctx.fillRect(p.x - 18, p.y - 3, 36, 6);
        const pulse = 4 + Math.sin(this.time * 8) * 3;
        ctx.fillStyle = 'rgba(255,183,74,.22)'; ctx.beginPath(); ctx.arc(p.x, p.y - 22, 30 + pulse, 0, Math.PI * 2); ctx.fill();
        ctx.fillStyle = '#ffcf66'; ctx.beginPath(); ctx.moveTo(p.x, p.y - 48 - pulse); ctx.quadraticCurveTo(p.x + 20, p.y - 18, p.x, p.y - 8); ctx.quadraticCurveTo(p.x - 16, p.y - 20, p.x, p.y - 48 - pulse); ctx.fill();
      }
      if (prop.type === 'well') {
        ctx.strokeStyle = '#899aa2'; ctx.lineWidth = 8; ctx.beginPath(); ctx.ellipse(p.x, p.y - 10, 28, 14, 0, 0, Math.PI * 2); ctx.stroke(); ctx.lineWidth = 1;
      }
      if (prop.type === 'crystal') {
        ctx.fillStyle = 'rgba(108,226,255,.18)'; ctx.beginPath(); ctx.arc(p.x, p.y - 26, 34, 0, Math.PI * 2); ctx.fill();
        ctx.fillStyle = '#8cecff'; ctx.beginPath(); ctx.moveTo(p.x, p.y - 76); ctx.lineTo(p.x + 18, p.y - 18); ctx.lineTo(p.x, p.y); ctx.lineTo(p.x - 15, p.y - 18); ctx.closePath(); ctx.fill();
      }
      if (prop.type === 'statue' || prop.type === 'altar') {
        ctx.fillStyle = '#577083'; ctx.fillRect(p.x - 30, p.y - 18, 60, 20);
        ctx.fillStyle = '#6f8ea2'; ctx.fillRect(p.x - 17, p.y - 85, 34, 70);
        ctx.fillStyle = '#91b8cd'; ctx.beginPath(); ctx.arc(p.x, p.y - 98, 18, 0, Math.PI * 2); ctx.fill();
      }
    }

    drawSeal(seal) {
      const ctx = this.ctx;
      const p = this.worldToScreen(seal.x, seal.y);
      const done = this.state.quests.seals[seal.index];
      const active = this.state.quests.activeSeal === seal.index;
      const pulse = 1 + Math.sin(this.time * 4 + seal.index) * 0.08;
      ctx.save(); ctx.translate(p.x, p.y - 6); ctx.scale(pulse, pulse);
      ctx.fillStyle = done ? 'rgba(100,255,202,.14)' : active ? 'rgba(255,152,180,.18)' : 'rgba(105,225,255,.14)';
      ctx.strokeStyle = done ? '#89f0bd' : active ? '#ff8aa2' : '#8cecff'; ctx.lineWidth = 3;
      ctx.beginPath();
      for (let i = 0; i < 6; i += 1) {
        const angle = -Math.PI / 2 + i * Math.PI / 3;
        const x = Math.cos(angle) * 40; const y = Math.sin(angle) * 22;
        if (i === 0) ctx.moveTo(x, y); else ctx.lineTo(x, y);
      }
      ctx.closePath(); ctx.fill(); ctx.stroke();
      ctx.fillStyle = '#dffbff'; ctx.font = '12px serif'; ctx.textAlign = 'center'; ctx.fillText(done ? '定' : ['西', '东', '北'][seal.index], 0, 4);
      ctx.restore();
    }

    drawPickup(pickup) {
      const ctx = this.ctx;
      const p = this.worldToScreen(pickup.x, pickup.y, 12 + Math.sin(pickup.bob) * 5);
      const colors = { shard: '#7fe9ff', relic: '#ffd993', potion: '#ff6f8c', equipment: '#b9a0ff' };
      ctx.fillStyle = `${colors[pickup.type]}33`; ctx.beginPath(); ctx.arc(p.x, p.y, 22, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = colors[pickup.type];
      if (pickup.type === 'potion') { ctx.fillRect(p.x - 7, p.y - 12, 14, 20); ctx.fillRect(p.x - 4, p.y - 17, 8, 5); }
      else { ctx.beginPath(); ctx.moveTo(p.x, p.y - 17); ctx.lineTo(p.x + 12, p.y); ctx.lineTo(p.x, p.y + 17); ctx.lineTo(p.x - 12, p.y); ctx.closePath(); ctx.fill(); }
    }

    drawActor(actor, kind) {
      const ctx = this.ctx;
      const p = this.worldToScreen(actor.x, actor.y);
      const radius = actor.radius;
      ctx.fillStyle = 'rgba(0,0,0,.26)'; ctx.beginPath(); ctx.ellipse(p.x, p.y + 8, radius * 1.05, radius * .45, 0, 0, Math.PI * 2); ctx.fill();
      let color = '#7adfff';
      if (kind === 'enemy' || kind === 'boss') color = actor.color;
      if (kind === 'npc') color = actor.color;
      if (kind === 'companion') color = '#e7c477';
      const bodyHeight = kind === 'boss' ? 82 : 48;
      ctx.fillStyle = color;
      ctx.beginPath();
      ctx.moveTo(p.x, p.y - bodyHeight);
      ctx.lineTo(p.x + radius * .7, p.y - 12);
      ctx.lineTo(p.x + radius * .45, p.y + 3);
      ctx.lineTo(p.x - radius * .45, p.y + 3);
      ctx.lineTo(p.x - radius * .7, p.y - 12);
      ctx.closePath(); ctx.fill();
      ctx.fillStyle = kind === 'enemy' || kind === 'boss' ? '#d8ecf5' : '#f4e4d2';
      ctx.beginPath(); ctx.arc(p.x, p.y - bodyHeight - 8, kind === 'boss' ? 18 : 11, 0, Math.PI * 2); ctx.fill();
      if (kind === 'player') {
        ctx.strokeStyle = '#dffbff'; ctx.lineWidth = 3; ctx.beginPath(); ctx.moveTo(p.x + 5, p.y - 32); ctx.lineTo(p.x + 25, p.y - 58); ctx.stroke(); ctx.lineWidth = 1;
        if (this.player.guardTimer > 0) { ctx.strokeStyle = 'rgba(155,239,255,.75)'; ctx.lineWidth = 2; ctx.beginPath(); ctx.arc(p.x, p.y - 28, 35 + Math.sin(this.time * 5) * 3, 0, Math.PI * 2); ctx.stroke(); ctx.lineWidth = 1; }
      }
      if (kind === 'companion') {
        ctx.strokeStyle = '#f8e0a0'; ctx.beginPath(); ctx.arc(p.x + 14, p.y - 38, 13, -1.1, 1.1); ctx.stroke();
      }
      if (kind === 'npc') {
        ctx.fillStyle = 'rgba(4,14,22,.82)'; ctx.fillRect(p.x - 48, p.y - bodyHeight - 42, 96, 19);
        ctx.fillStyle = '#eaf8ff'; ctx.font = '11px sans-serif'; ctx.textAlign = 'center'; ctx.fillText(actor.name, p.x, p.y - bodyHeight - 28);
      }
      if (kind === 'enemy' || kind === 'boss' || kind === 'companion') {
        const ratio = clamp(actor.hp / actor.maxHp, 0, 1);
        const bw = kind === 'boss' ? 86 : 48;
        ctx.fillStyle = 'rgba(0,0,0,.6)'; ctx.fillRect(p.x - bw / 2, p.y - bodyHeight - 38, bw, 6);
        ctx.fillStyle = kind === 'companion' ? '#eacb71' : '#ff607c'; ctx.fillRect(p.x - bw / 2, p.y - bodyHeight - 38, bw * ratio, 6);
        if (actor.elite) { ctx.fillStyle = '#ffe08a'; ctx.font = '10px sans-serif'; ctx.textAlign = 'center'; ctx.fillText('精英', p.x, p.y - bodyHeight - 43); }
      }
    }

    drawWall(wall) {
      const ctx = this.ctx;
      for (let i = 0; i < 6; i += 1) {
        const x = wall.x + (wall.w / 5) * i;
        const y = wall.y + wall.h / 2;
        const p = this.worldToScreen(x, y);
        const alpha = clamp(wall.life / 1.2, 0.25, 1);
        ctx.fillStyle = `rgba(154,239,255,${alpha * .75})`;
        ctx.beginPath(); ctx.moveTo(p.x, p.y - 58); ctx.lineTo(p.x + 17, p.y); ctx.lineTo(p.x - 15, p.y); ctx.closePath(); ctx.fill();
      }
    }

    drawProjectile(projectile) {
      const ctx = this.ctx;
      const p = this.worldToScreen(projectile.x, projectile.y, 20);
      ctx.fillStyle = `${projectile.color}33`; ctx.beginPath(); ctx.arc(p.x, p.y, projectile.radius * 2, 0, Math.PI * 2); ctx.fill();
      ctx.fillStyle = projectile.color; ctx.beginPath(); ctx.arc(p.x, p.y, projectile.radius, 0, Math.PI * 2); ctx.fill();
    }

    drawEffect(effect) {
      const ctx = this.ctx;
