  class Input {
    constructor(canvas) {
      this.canvas = canvas;
      this.keys = new Set();
      this.pressed = new Set();
      this.pointer = { x: W / 2, y: H / 2, clicked: false, rightClicked: false };

      window.addEventListener('keydown', (event) => {
        const rawKey = event.key.toLowerCase();
        const key = this.normalizeKey(rawKey);
        if (!this.keys.has(key)) this.pressed.add(key);
        this.keys.add(key);
        if (['w', 'a', 's', 'd', ' ', '1', '2', '3', '4', 'q', 'i', 'e', 'r', 'escape', 'enter',
          'arrowup', 'arrowdown', 'arrowleft', 'arrowright'].includes(rawKey)) {
          event.preventDefault();
        }
      });
      window.addEventListener('keyup', (event) => this.keys.delete(this.normalizeKey(event.key.toLowerCase())));

      canvas.addEventListener('pointermove', (event) => this.updatePointer(event));
      canvas.addEventListener('pointerdown', (event) => {
        this.updatePointer(event);
        if (event.button === 0) this.pointer.clicked = true;
        if (event.button === 2) this.pointer.rightClicked = true;
      });
      canvas.addEventListener('contextmenu', (event) => event.preventDefault());
    }

    normalizeKey(key) {
      return ({ arrowup: 'w', arrowdown: 's', arrowleft: 'a', arrowright: 'd' })[key] || key;
    }

    updatePointer(event) {
      const rect = this.canvas.getBoundingClientRect();
      this.pointer.x = ((event.clientX - rect.left) / rect.width) * W;
      this.pointer.y = ((event.clientY - rect.top) / rect.height) * H;
    }

    down(key) { return this.keys.has(key); }
    wasPressed(key) { return this.pressed.has(key); }
    endFrame() {
      this.pressed.clear();
      this.pointer.clicked = false;
      this.pointer.rightClicked = false;
    }
  }

  class SoundBank {
    constructor() { this.context = null; this.enabled = true; }
    ensure() {
      if (!this.enabled) return null;
      if (!this.context) {
        const AudioCtx = window.AudioContext || window.webkitAudioContext;
        if (AudioCtx) this.context = new AudioCtx();
      }
      return this.context;
    }
    tone(frequency = 440, duration = 0.07, type = 'sine', volume = 0.025) {
      const ctx = this.ensure();
      if (!ctx) return;
      const oscillator = ctx.createOscillator();
      const gain = ctx.createGain();
      oscillator.type = type;
      oscillator.frequency.value = frequency;
      gain.gain.setValueAtTime(volume, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.0001, ctx.currentTime + duration);
      oscillator.connect(gain).connect(ctx.destination);
      oscillator.start();
      oscillator.stop(ctx.currentTime + duration);
    }
  }

  class Projectile extends Entity {
    constructor(options) {
      super(options.x, options.y, options.radius || 9);
      Object.assign(this, {
        vx: options.vx,
        vy: options.vy,
        damage: options.damage,
        owner: options.owner,
        life: options.life || 2,
        color: options.color || '#9cecff',
        slow: options.slow || 0,
        pierce: options.pierce || 0,
        mage: options.mage || false,
      });
    }
    update(game, dt) {
      this.life -= dt;
      if (this.life <= 0) { this.dead = true; return; }
      const nx = this.x + this.vx * dt;
      const ny = this.y + this.vy * dt;
      if (game.isBlocked(nx, ny, this.radius)) { this.dead = true; return; }
      this.x = nx;
      this.y = ny;

      const targets = this.owner === 'player'
        ? game.enemies.filter((enemy) => !enemy.dead)
        : [game.player, game.companion].filter(Boolean).filter((target) => !target.dead);
      for (const target of targets) {
        if (distance(this, target) <= this.radius + target.radius) {
          game.dealDamage(target, this.damage, this.owner, { slow: this.slow, mage: this.mage });
          if (this.pierce > 0) this.pierce -= 1;
          else this.dead = true;
          break;
        }
      }
    }
  }
