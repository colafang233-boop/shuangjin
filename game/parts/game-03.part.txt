  class Game {
    constructor() {
      this.canvas = document.getElementById('game');
      this.ctx = this.canvas.getContext('2d');
      this.input = new Input(this.canvas);
      this.sound = new SoundBank();
      this.state = createDefaultState();
      this.player = new Player(this.state.player);
      this.scene = null;
      this.enemies = [];
      this.npcs = [];
      this.pickups = [];
      this.projectiles = [];
      this.effects = [];
      this.walls = [];
      this.companion = null;
      this.boss = null;
      this.currentInteractable = null;
      this.mode = 'menu';
      this.time = 0;
      this.lastTime = performance.now();
      this.autosaveTimer = 0;
      this.camera = { x: 0, y: 0 };
      this.shake = 0;
      this.dialogueAction = null;
      this.interactionRepeatCooldown = 0;
      this.dialogueRepeatCooldown = 0;
      this.bindUI();
      this.drawMenuBackdrop();
      requestAnimationFrame((t) => this.loop(t));
    }

    bindUI() {
      this.ui = {
        start: document.getElementById('start-screen'), hud: document.getElementById('hud'),
        ending: document.getElementById('ending-screen'), dialogue: document.getElementById('dialogue'),
        speaker: document.getElementById('speaker'), dialogueText: document.getElementById('dialogue-text'),
        dialogueActions: document.getElementById('dialogue-actions'), panel: document.getElementById('side-panel'),
        panelTitle: document.getElementById('panel-title'), panelContent: document.getElementById('panel-content'),
        context: document.getElementById('context-prompt'), messageFeed: document.getElementById('message-feed'),
        hpBar: document.getElementById('hp-bar'), hpText: document.getElementById('hp-text'),
        energyBar: document.getElementById('energy-bar'), energyText: document.getElementById('energy-text'),
        xpBar: document.getElementById('xp-bar'), levelText: document.getElementById('level-text'),
        potionCount: document.getElementById('potion-count'), questTracker: document.getElementById('quest-tracker'),
        sceneName: document.getElementById('scene-name'), saveIndicator: document.getElementById('save-indicator'),
        bossHud: document.getElementById('boss-hud'), bossName: document.getElementById('boss-name'),
        bossPhase: document.getElementById('boss-phase'), bossBar: document.getElementById('boss-bar-fill'),
        endingKicker: document.getElementById('ending-kicker'), endingTitle: document.getElementById('ending-title'),
        endingText: document.getElementById('ending-text'), endingSummary: document.getElementById('ending-summary'),
        continueButton: document.getElementById('continue-game'),
      };

      document.getElementById('new-game').addEventListener('click', () => this.newGame());
      this.ui.continueButton.addEventListener('click', () => this.continueGame());
      document.getElementById('restart-game').addEventListener('click', () => this.newGame());
      document.getElementById('return-menu').addEventListener('click', () => this.returnToMenu());
      document.getElementById('quest-button').addEventListener('click', () => this.togglePanel('quests'));
      document.getElementById('inventory-button').addEventListener('click', () => this.togglePanel('inventory'));
      document.getElementById('close-panel').addEventListener('click', () => this.closePanel());
      document.querySelectorAll('.skill').forEach((button) => {
        button.addEventListener('click', () => {
          if (this.mode === 'playing') this.player.useSkill(this, Number(button.dataset.skill));
        });
      });
      this.ui.continueButton.disabled = !this.hasSave();
    }

    hasSave() {
      try { return Boolean(localStorage.getItem(SAVE_KEY)); } catch { return false; }
    }

    newGame() {
      try { localStorage.removeItem(SAVE_KEY); } catch { /* no-op */ }
      this.state = createDefaultState();
      this.player = new Player(this.state.player);
      this.mode = 'playing';
      this.ui.start.classList.add('hidden');
      this.ui.ending.classList.add('hidden');
      this.ui.hud.classList.remove('hidden');
      this.loadScene('village');
      this.message('在村中寻找韩伯，调查失踪事件。', 'good');
      this.showDialogue('凌霜', '北境的风里带着不属于人间的寒意。师父留下的最后线索，就在这座村庄。', [
        { label: '进入寒霜村', action: () => {} },
      ]);
    }

    continueGame() {
      try {
        const saved = JSON.parse(localStorage.getItem(SAVE_KEY));
        if (!saved || saved.version !== 1) throw new Error('invalid save');
        this.state = saved;
        this.state.world ||= { collectedPickups: [] };
        this.state.world.collectedPickups ||= [];
        this.player = new Player(this.state.player);
        this.mode = 'playing';
        this.ui.start.classList.add('hidden');
        this.ui.ending.classList.add('hidden');
        this.ui.hud.classList.remove('hidden');
        this.loadScene(this.state.scene || 'village', { preservePosition: true });
        this.message('已读取自动存档。', 'good');
      } catch {
        this.message('存档无法读取，已开始新游戏。', 'bad');
        this.newGame();
      }
    }

    returnToMenu() {
      this.mode = 'menu';
      this.ui.ending.classList.add('hidden');
      this.ui.hud.classList.add('hidden');
      this.ui.start.classList.remove('hidden');
      this.ui.continueButton.disabled = !this.hasSave();
      this.drawMenuBackdrop();
    }

      this.state.player.x = this.player.x;
      this.state.player.y = this.player.y;
      try {
        localStorage.setItem(SAVE_KEY, JSON.stringify(this.state));
        this.ui.saveIndicator.classList.add('show');
